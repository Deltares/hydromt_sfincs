    @field_validator("tref", "tstart", "tstop", mode="before")
    @classmethod
    def parse_custom_datetime(cls, v):
        if isinstance(v, str):
            try:
                return datetime.strptime(v.strip(), "%Y%m%d %H%M%S")
            except ValueError:
                raise ValueError(
                    f"Invalid datetime format: {v}. Expected format: YYYYMMDD HHMMSS"
                )
        return v
